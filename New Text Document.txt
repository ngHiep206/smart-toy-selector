    <script>
        // Cấu hình Gemini API
        const apiKey = "https://github.com/ngHiep206/smart-toy-selector.git"; // Được cung cấp bởi môi trường thực thi

        // Hàm gọi Gemini API có cơ chế retry (Exponential Backoff)
        async function fetchGeminiAPI(promptText) {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: promptText }] }]
            };

            const delays = [1000, 2000, 4000, 8000, 16000];
            for (let attempt = 0; attempt <= delays.length; attempt++) {
                try {
                    const response = await fetch(url, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    
                    if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
                    
                    const result = await response.json();
                    const text = result.candidates?.[0]?.content?.parts?.[0]?.text;
                    if(text) return text;
                    throw new Error("Empty response from AI");
                } catch (error) {
                    if (attempt === delays.length) {
                        return "Rất tiếc, hệ thống AI đang bận. Vui lòng thử lại sau nhé!";
                    }
                    await new Promise(resolve => setTimeout(resolve, delays[attempt]));
                }
            }
        }

        // UI Toast Helper (Thay thế cho alert)
        function showToast(message) {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = 'bg-gray-900 text-white px-6 py-3 rounded-xl shadow-2xl flex items-center gap-3 toast-enter-active';
            toast.innerHTML = `<i class="fa-solid fa-circle-check text-green-400"></i> <span class="font-semibold">${message}</span>`;
            
            container.appendChild(toast);
            
            setTimeout(() => {
                toast.classList.remove('toast-enter-active');
                toast.classList.add('toast-leave-active');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }

        // TÍNH NĂNG AI: TẠO CÂU CHUYỆN / KỊCH BẢN CHƠI TRONG MODAL
        async function generateAIStory(toyName) {
            const btn = document.getElementById('btn-generate-story');
            const resultDiv = document.getElementById('ai-story-result');
            
            // UI Loading state
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-2"></i> AI đang sáng tác...`;
            resultDiv.classList.remove('hidden');
            resultDiv.innerHTML = `<div class="h-16 rounded-lg shimmer w-full"></div>`;

            // Prompt cho Gemini
            const prompt = `Hãy sáng tạo một kịch bản trò chơi tương tác ngắn hoặc một câu chuyện kể cực kỳ thú vị (dưới 100 chữ) để ba mẹ chơi cùng bé 1 tuổi với món đồ chơi: "${toyName}". Yêu cầu: Vui nhộn, kích thích trí tưởng tượng, có kèm một vài lời thoại mẫu hoặc âm thanh vui nhộn (như bùm, chéo, meo meo...) để ba mẹ bắt chước nói với bé. Đóng vai chuyên gia, bắt đầu thẳng vào nội dung.`;

            // Gọi API
            const storyText = await fetchGeminiAPI(prompt);
            
            // Xử lý xuống dòng HTML cơ bản
            const formattedText = storyText.replace(/\n/g, '<br>');

            // Hiển thị kết quả
            resultDiv.innerHTML = `<p class="italic">"${formattedText}"</p>`;
            btn.innerHTML = `✨ Tạo kịch bản khác`;
            btn.disabled = false;
        }

        // Hàm Đóng Modal
        function closeModal() {
            document.getElementById('toy-modal').classList.add('hidden');
        }

        document.getElementById('toy-modal').addEventListener('click', function(e) {
            if (e.target === this) {
                closeModal();
            }
        });

        // Hàm Add To Cart dùng Toast
        function addToCart(toyName) {
            closeModal();
            showToast(`Đã thêm <b>${toyName}</b> vào giỏ hàng!`);
        }
